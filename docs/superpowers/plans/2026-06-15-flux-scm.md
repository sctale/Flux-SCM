# Flux-SCM 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 构建面向小批量多品种采购的物料与供应商管理 Windows 桌面平台，以卡拉杰克矩阵为核心实现供应商科学分类与差异化管控。

**Architecture:** Electron + React 单体桌面应用，主进程管理窗口与文件系统，渲染进程承载 React UI，通过 IPC Bridge 调用 SQLite 数据库。数据库使用 better-sqlite3 嵌入式方案，ORM 使用 Drizzle。

**Tech Stack:** Electron 33+ / React 18 / TypeScript / Ant Design 5 / ECharts 5 / Zustand / Drizzle ORM / better-sqlite3 / Vite

```typescript
  });
```

```typescript
    mainWindow.loadURL('http://localhost:5173');
  } else {
    mainWindow.loadFile(path.join(__dirname, '../dist/renderer/index.html'));
  }

app.whenReady().then(createWindow);

app.on('window-all-closed', () => {
  if (process.platform !== 'darwin') app.quit();

  if (BrowserWindow.getAllWindows().length === 0) createWindow();

// Window control IPC
ipcMain.on('window:minimize', () => mainWindow?.minimize());
ipcMain.on('window:maximize', () => {
```

```typescript
// electron/database/index.ts
import Database from 'better-sqlite3';
import { drizzle } from 'drizzle-orm/better-sqlite3';
import * as schema from './schema';
import path from 'path';
import { app } from 'electron';
import fs from 'fs';

let db: ReturnType<typeof drizzle> | null = null;

export function getDatabase() {
  if (db) return db;

  const userDataPath = app.getPath('userData');
  const dbPath = path.join(userDataPath, 'flux-scm.db');

  // Ensure directory exists
  const dir = path.dirname(dbPath);
  if (!fs.existsSync(dir)) fs.mkdirSync(dir, { recursive: true });

  const sqlite = new Database(dbPath);
  sqlite.pragma('journal_mode = WAL');
  sqlite.pragma('foreign_keys = ON');

  db = drizzle(sqlite, { schema });

  // Run migrations (simple approach for SQLite)
  initializeTables(sqlite);

  return db;
}

function initializeTables(sqlite: Database.Database) {
  const tables = [
    `CREATE TABLE IF NOT EXISTS suppliers (
      id TEXT PRIMARY KEY, name TEXT NOT NULL, short_name TEXT,
      unified_code TEXT, category TEXT, kraljic_quadrant TEXT,
      profit_impact_score REAL DEFAULT 0, supply_risk_score REAL DEFAULT 0,
      performance_grade TEXT, overall_score REAL DEFAULT 0,
      quality_score REAL DEFAULT 0, cost_score REAL DEFAULT 0,
      delivery_score REAL DEFAULT 0, service_score REAL DEFAULT 0,
      status TEXT NOT NULL DEFAULT 'pending',
      address TEXT, contact_person TEXT, contact_phone TEXT, contact_email TEXT,
      bank_name TEXT, bank_account TEXT, remark TEXT,
      created_at TEXT NOT NULL, updated_at TEXT NOT NULL
    )`,
    `CREATE TABLE IF NOT EXISTS supplier_qualifications (
      id TEXT PRIMARY KEY, supplier_id TEXT NOT NULL REFERENCES suppliers(id),
      cert_type TEXT NOT NULL, cert_number TEXT,
      issue_date TEXT, expiry_date TEXT, file_path TEXT,
      status TEXT NOT NULL DEFAULT 'valid', remark TEXT,
      created_at TEXT NOT NULL, updated_at TEXT NOT NULL
    )`,
    `CREATE TABLE IF NOT EXISTS materials (
      id TEXT PRIMARY KEY, code TEXT NOT NULL UNIQUE,
      name TEXT NOT NULL, specification TEXT, unit TEXT NOT NULL,
      category TEXT, material_type TEXT,
      safety_stock REAL DEFAULT 0, lead_time INTEGER DEFAULT 0,
      drawing_no TEXT, remark TEXT,
      created_at TEXT NOT NULL, updated_at TEXT NOT NULL
    )`,
    `CREATE TABLE IF NOT EXISTS material_suppliers (
      id TEXT PRIMARY KEY, material_id TEXT NOT NULL REFERENCES materials(id),
      supplier_id TEXT NOT NULL REFERENCES suppliers(id),
      unit_price REAL, min_order_qty INTEGER DEFAULT 1,
      lead_time_days INTEGER, is_preferred INTEGER DEFAULT 0,
      last_quote_date TEXT, remark TEXT,
      created_at TEXT NOT NULL, updated_at TEXT NOT NULL
    )`,
    `CREATE TABLE IF NOT EXISTS purchase_orders (
      id TEXT PRIMARY KEY, order_no TEXT NOT NULL UNIQUE,
      supplier_id TEXT NOT NULL REFERENCES suppliers(id),
      title TEXT, status TEXT NOT NULL DEFAULT 'draft',
      total_amount REAL DEFAULT 0, order_date TEXT,
      expected_date TEXT, payment_term TEXT,
      remark TEXT, created_by TEXT,
      created_at TEXT NOT NULL, updated_at TEXT NOT NULL
    )`,
    `CREATE TABLE IF NOT EXISTS order_items (
      id TEXT PRIMARY KEY, order_id TEXT NOT NULL REFERENCES purchase_orders(id),
      material_id TEXT NOT NULL REFERENCES materials(id),
      quantity REAL NOT NULL, unit_price REAL NOT NULL,
      amount REAL NOT NULL, delivered_qty REAL DEFAULT 0,
      qualified_qty REAL DEFAULT 0, expected_date TEXT, remark TEXT
    )`,
    `CREATE TABLE IF NOT EXISTS deliveries (
      id TEXT PRIMARY KEY, order_id TEXT NOT NULL REFERENCES purchase_orders(id),
      order_item_id TEXT NOT NULL REFERENCES order_items(id),
      delivery_date TEXT NOT NULL, quantity REAL NOT NULL,
      qualified_qty REAL DEFAULT 0, inspection_result TEXT,
      inspection_remark TEXT, created_at TEXT NOT NULL
    )`,
    `CREATE TABLE IF NOT EXISTS risk_alerts (
      id TEXT PRIMARY KEY, supplier_id TEXT REFERENCES suppliers(id),
      alert_type TEXT NOT NULL, level TEXT NOT NULL,
      title TEXT NOT NULL, description TEXT,
      status TEXT NOT NULL DEFAULT 'active',
      resolved_at TEXT, created_at TEXT NOT NULL
    )`,
    `CREATE TABLE IF NOT EXISTS supplier_contacts (
      id TEXT PRIMARY KEY, supplier_id TEXT NOT NULL REFERENCES suppliers(id),
      name TEXT NOT NULL, position TEXT, phone TEXT, email TEXT,
      is_primary INTEGER DEFAULT 0, remark TEXT
    )`,
  ];

  const transaction = sqlite.transaction(() => {
    for (const sql of tables) {
      sqlite.exec(sql);
    }
  });
  transaction();
}

export { schema };
```

```typescript
interface SupplyRiskInput {
  alternativeCount: number;       // 替代供应商数量
  switchingCost: number;          // 转换成本 1-10
  replacementLeadTime: number;    // 替换周期(天)
  productComplexity: number;      // 产品复杂度 1-10
}

  name: string;

/**
 * 计算利润影响分数 (0-100)
 * @param annualSpend 该供应商/品类的年采购金额
 * @param totalSpend 年度总采购金额
 */
export function calculateProfitImpact(annualSpend: number, totalSpend: number): number {
  if (totalSpend === 0) return 0;
  const ratio = annualSpend / totalSpend;
  // 使用对数缩放使分布更合理
  const score = Math.min(100, Math.max(0, ratio * 100 * 5 + Math.log10(ratio * 100 + 1) * 20));
  return Math.round(score * 10) / 10;
}

/**
 * 计算供应风险分数 (0-100)
 */
export function calculateSupplyRisk(input: SupplyRiskInput): number {
  const { alternativeCount, switchingCost, replacementLeadTime, productComplexity } = input;

  // 供应商数量评分 (0-25): 越少风险越高
  const countScore = alternativeCount <= 1 ? 25 : alternativeCount <= 3 ? 20 : alternativeCount <= 10 ? 10 : 3;

  // 转换成本评分 (0-25)
  const switchScore = (switchingCost / 10) * 25;

  // 替换周期评分 (0-25): 越长风险越高
  const leadScore = replacementLeadTime <= 7 ? 3 : replacementLeadTime <= 30 ? 10 : replacementLeadTime <= 90 ? 18 : 25;

  // 产品复杂度评分 (0-25)
  const complexScore = (productComplexity / 10) * 25;

  return Math.round(Math.min(100, countScore + switchScore + leadScore + complexScore) * 10) / 10;
}
```

```typescript
// src/utils/scoring.ts
import type { PerformanceGrade } from '@/types/common';

interface QCDSInput {
  qualityScore: number;   // 0-100
  costScore: number;      // 0-100
  deliveryScore: number;  // 0-100
  serviceScore: number;   // 0-100
}

interface QCDSWeights {
  quality: number;   // default 0.30
  cost: number;      // default 0.25
  delivery: number;  // default 0.30
  service: number;   // default 0.15
}

interface QCDSResult {
  overallScore: number;
  grade: PerformanceGrade;
  weights: QCDSWeights;
}

const DEFAULT_WEIGHTS: QCDSWeights = {
  quality: 0.30,
  cost: 0.25,
  delivery: 0.30,
  service: 0.15,
};

/**
 * 计算 QCDS 加权综合评分
 */
export function calculateQCDS(
  input: QCDSInput,
  weights: Partial<QCDSWeights> = {}
): QCDSResult {
  const w = { ...DEFAULT_WEIGHTS, ...weights };
  const overallScore =
    input.qualityScore * w.quality +
    input.costScore * w.cost +
    input.deliveryScore * w.delivery +
    input.serviceScore * w.service;

  const roundedScore = Math.round(overallScore * 10) / 10;

  return {
    overallScore: roundedScore,
    grade: getPerformanceGrade(roundedScore),
    weights: w,
  };
}

/**
 * 根据分数确定绩效等级
 */
export function getPerformanceGrade(score: number): PerformanceGrade {
  if (score >= 90) return 'A';
  if (score >= 80) return 'B';
  if (score >= 70) return 'C';
  return 'D';
}

/**
 * 根据到货记录计算交付分数
 */
export function calculateDeliveryScore(
  totalOrders: number,
  onTimeOrders: number
): number {
  if (totalOrders === 0) return 100;
  return Math.round((onTimeOrders / totalOrders) * 100);
}

/**
 * 根据来料检验记录计算质量分数
 */
export function calculateQualityScore(
  totalDeliveries: number,
  qualifiedDeliveries: number
): number {
  if (totalDeliveries === 0) return 100;
  return Math.round((qualifiedDeliveries / totalDeliveries) * 100);
}

/**
 * 绩效等级颜色映射
 */
export const gradeColors: Record<PerformanceGrade, string> = {
  A: '#52C41A',
  B: '#1677FF',
  C: '#FAAD14',
  D: '#FF4D4F',
};
```

```typescript
import React, { useState } from 'react';
import {
  Table, Button, Input, Select, Space, Tag, Tooltip, Card, Row, Col, Statistic,
} from 'antd';
import {
  PlusOutlined, ExportOutlined, SearchOutlined, FilterOutlined,
} from '@ant-design/icons';
import type { Supplier } from '@/types/supplier';
import { quadrantLabels, quadrantColors } from '@/utils/kraljic';
import { gradeColors } from '@/utils/scoring';

const { Option } = Select;

interface SupplierListProps {
  suppliers: Supplier[];
  onAdd: () => void;
  onEdit: (id: string) => void;
  onDelete: (id: string) => void;
  onView: (id: string) => void;
}

const statusMap: Record<string, { color: string; text: string }> = {
  pending: { color: 'default', text: '待审核' },
  trial: { color: 'processing', text: '试用中' },
  active: { color: 'success', text: '合格' },
  suspended: { color: 'warning', text: '暂停' },
  blacklisted: { color: 'error', text: '黑名单' },
};

const SupplierList: React.FC<SupplierListProps> = ({
  suppliers, onAdd, onEdit, onDelete, onView,

  const [filterQuadrant, setFilterQuadrant] = useState<string | undefined>();
  const [filterGrade, setFilterGrade] = useState<string | undefined>();
  const [filterStatus, setFilterStatus] = useState<string | undefined>();

  const filteredData = suppliers.filter((s) => {
    const matchSearch = !searchText ||
      s.name.includes(searchText) ||
      s.shortName?.includes(searchText) ||
      s.unifiedCode?.includes(searchText);
    const matchQuadrant = !filterQuadrant || s.kraljicQuadrant === filterQuadrant;
    const matchGrade = !filterGrade || s.performanceGrade === filterGrade;
    const matchStatus = !filterStatus || s.status === filterStatus;
    return matchSearch && matchQuadrant && matchGrade && matchStatus;
  });

  // 统计数据
  const stats = {
    total: suppliers.length,
    active: suppliers.filter(s => s.status === 'active').length,
    strategic: suppliers.filter(s => s.kraljicQuadrant === 'strategic').length,
    riskCount: suppliers.filter(s => s.performanceGrade === 'D').length,
  };

  const columns = [
    {
      title: '供应商名称',
      dataIndex: 'name',
      key: 'name',
      render: (text: string, record: Supplier) => (
        <a onClick={() => onView(record.id)} style={{ fontWeight: 500 }}>{text}</a>
      ),
    },
    {
      title: '矩阵分类',
      dataIndex: 'kraljicQuadrant',
      key: 'kraljicQuadrant',
      width: 100,
      render: (val: string) => val ? (
        <Tag color={quadrantColors[val as keyof typeof quadrantColors]}>
          {quadrantLabels[val as keyof typeof quadrantLabels]}
        </Tag>
      ) : '-',
    },
    {
      title: '绩效等级',
      dataIndex: 'performanceGrade',
      key: 'performanceGrade',
      width: 80,
      render: (val: string) => val ? (
        <Tag color={gradeColors[val as keyof typeof gradeColors]}>{val}</Tag>
      ) : '-',
    },
    {
      title: '综合评分',
      dataIndex: 'overallScore',
      key: 'overallScore',
      width: 90,
      sorter: (a: Supplier, b: Supplier) => a.overallScore - b.overallScore,
      render: (val: number) => val > 0 ? val.toFixed(1) : '-',
    },
    {
      title: '产品类别',
      dataIndex: 'category',
      key: 'category',
      width: 100,
    },

      title: '联系人',
      dataIndex: 'contactPerson',
      key: 'contactPerson',
      width: 90,
    },
    {
      title: '电话',
      dataIndex: 'contactPhone',
      key: 'contactPhone',
      width: 130,

    {
      title: '状态',
      dataIndex: 'status',
      key: 'status',
      width: 80,
      render: (val: string) => {
        const info = statusMap[val];
        return info ? <Tag color={info.color}>{info.text}</Tag> : val;
      },
    },
    {
      title: '操作',
      key: 'action',
      width: 150,
      render: (_: unknown, record: Supplier) => (
        <Space size="small">
          <Button type="link" size="small" onClick={() => onView(record.id)}>查看</Button>
          <Button type="link" size="small" onClick={() => onEdit(record.id)}>编辑</Button>
          <Button type="link" size="small" danger onClick={() => onDelete(record.id)}>删除</Button>
        </Space>
      ),
    },
  ];

  return (
    <div>
      {/* 统计卡片 */}
      <Row gutter={16} style={{ marginBottom: 16 }}>
        <Col span={6}>
          <Card size="small"><Statistic title="供应商总数" value={stats.total} /></Card>
        </Col>
        <Col span={6}>
          <Card size="small"><Statistic title="合格供应商" value={stats.active} valueStyle={{ color: '#52C41A' }} /></Card>
        </Col>
        <Col span={6}>
          <Card size="small"><Statistic title="战略型" value={stats.strategic} valueStyle={{ color: '#1677FF' }} /></Card>
        </Col>
        <Col span={6}>
          <Card size="small"><Statistic title="风险供应商" value={stats.riskCount} valueStyle={{ color: '#FF4D4F' }} /></Card>
        </Col>
      </Row>

      <Card size="small" style={{ marginBottom: 16 }}>

          <Input
            placeholder="搜索供应商名称/编码"
            prefix={<SearchOutlined />}
            value={searchText}
            onChange={e => setSearchText(e.target.value)}
            style={{ width: 250 }}

          <Select

            onChange={setFilterQuadrant}
```

```typescript
// src/components/material/MaterialForm.tsx
import React, { useEffect } from 'react';
import { Modal, Form, Input, InputNumber, Select, Row, Col } from 'antd';
import type { MaterialFormData } from '@/types/material';
import { categoryCodeMap, materialTypeCodeMap, generateMaterialCode } from '@/utils/coding';

interface MaterialFormProps {
  visible: boolean;
  onSubmit: (data: MaterialFormData) => void;
  onCancel: () => void;
}

  }, [visible, form]);

    // 自动生成编码

      materialType: materialTypeCode,
      specification: specCode,
    });

      <Form form={form} layout="vertical">
```

```typescript
// src/components/order/OrderList.tsx
import React, { useState } from 'react';
import { Table, Button, Input, Space, Tag, Card, Select } from 'antd';
import { PlusOutlined, SearchOutlined } from '@ant-design/icons';
import type { PurchaseOrder } from '@/types/order';

interface OrderListProps {
  orders: PurchaseOrder[];
  onAdd: () => void;
  onView: (id: string) => void;
}

const statusMap: Record<string, { color: string; text: string }> = {
  draft: { color: 'default', text: '草稿' },
  submitted: { color: 'processing', text: '已提交' },
  confirmed: { color: 'blue', text: '已确认' },
  partial_delivered: { color: 'cyan', text: '部分到货' },
  delivered: { color: 'success', text: '已到货' },
  closed: { color: 'default', text: '已关闭' },
  cancelled: { color: 'error', text: '已取消' },
};

const OrderList: React.FC<OrderListProps> = ({ orders, onAdd, onView }) => {
  const [search, setSearch] = useState('');
  const [filterStatus, setFilterStatus] = useState<string | undefined>();

  const filtered = orders.filter(o => {
    const matchSearch = !search || o.orderNo.includes(search) || o.title?.includes(search);
    const matchStatus = !filterStatus || o.status === filterStatus;
    return matchSearch && matchStatus;
  });

  const columns = [
    { title: '订单号', dataIndex: 'orderNo', key: 'orderNo', width: 140, render: (v: string) => <a>{v}</a> },
    { title: '订单标题', dataIndex: 'title', key: 'title', width: 200 },
    { title: '订单金额', dataIndex: 'totalAmount', key: 'totalAmount', width: 120, render: (v: number) => `¥${v?.toLocaleString() || 0}` },
    { title: '订单日期', dataIndex: 'orderDate', key: 'orderDate', width: 110 },
    { title: '期望交期', dataIndex: 'expectedDate', key: 'expectedDate', width: 110 },
    {
      title: '状态', dataIndex: 'status', key: 'status', width: 100,
      render: (v: string) => { const info = statusMap[v]; return info ? <Tag color={info.color}>{info.text}</Tag> : v; },
    },
    {
      title: '操作', key: 'action', width: 80,
      render: (_: unknown, record: PurchaseOrder) => <Button type="link" size="small" onClick={() => onView(record.id)}>查看</Button>,
    },
  ];

  return (
    <div>
      <Card size="small" style={{ marginBottom: 16 }}>
        <Space>
          <Input placeholder="搜索订单号/标题" prefix={<SearchOutlined />} value={search}
            onChange={e => setSearch(e.target.value)} style={{ width: 250 }} allowClear />
          <Select placeholder="订单状态" value={filterStatus} onChange={setFilterStatus}
            style={{ width: 120 }} allowClear>
            {Object.entries(statusMap).map(([k, v]) => <Select.Option key={k} value={k}>{v.text}</Select.Option>)}
          </Select>
          <Button type="primary" icon={<PlusOutlined />} onClick={onAdd}>新建订单</Button>
        </Space>
      </Card>
      <Table columns={columns} dataSource={filtered} rowKey="id" size="middle"
        pagination={{ pageSize: 15, showTotal: t => `共 ${t} 条` }} />
    </div>
  );
};

export default OrderList;
```
